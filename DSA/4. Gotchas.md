
1. Unordered_set uses hashing, and can’t be used to store vector / pairs, etc, as keys because they don’t have a hash. Either use regular set, or provide custom hash.
```
	1. unordered_set<pair<int,int>> zeros; → invalid
	2. set<pair<int,int>> zeros; → valid
```

2. “next_permutation()” returns false if passed the lexicographically largest possible element as argument.
3. find() returns a pointer when used without a smart for loop:
```
    map<char, int> m = {
        {'a', 1},
        {'b', 2},
        {'c', 3},
        {'d', 4}
    };

    for (const auto& [key, value]: m) {
        cout << key << ' ' << value << newl;        // smart for loop
    }

    map<char, int>::iterator it = m.find('a');      // use an iterator, which
    cout << it->first << ' ' << it->second << newl; // behaves like a pointer
```
4. Iterators cannot always be advanced with += , eg: for iterator to list element += overload is not defined, but advance() works.
```
    vector<int> v = {1, 2, 3, 4, 5};
    vector<int>::iterator itv = v.begin();
    itv += 2;                                    // works
    cout << *itv << endl;
    
    list<int> lst; 
    lst.assign(5, 10);
    list<int>::iterator itl = lst.begin();
    advance(itl, 2);                             // works
    // itl += 2;                                 // does NOT work
    cout << *itl << endl;
```
